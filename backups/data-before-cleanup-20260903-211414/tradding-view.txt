indicator("محرك القرار المؤسسي V56 - Ultimate Transparency", overlay=true, max_lines_count=500, max_labels_count=500, max_boxes_count=500)

// ==========================================
// 1. الإعدادات
// ==========================================
group_market = "اختر السوق"
market_type = input.string("تلقائي (Auto)", "نوع السوق", options=["تلقائي (Auto)", "البورصة المصرية (EGX)", "السعودية (Tadawul)", "العملات الرقمية (Crypto)"], group=group_market)
string active_market = market_type == "تلقائي (Auto)" ? (syminfo.type == "crypto" ? "Crypto" : syminfo.currency == "EGP" ? "EGX" : "Global") : market_type
benchmark_ticker = input.symbol("SPY", "رمز المؤشر العام (غيّره لـ EGX:EGX30 أو اتركه فارغاً)", group=group_market)

group_capital = "💰 إدارة رأس المال"
account_balance = input.float(10000, "حجم رأس المال", group=group_capital)
risk_per_trade = input.float(1.0, "نسبة المخاطرة %", group=group_capital)

group_ui = "🎨 الشكل والرسم"
show_zones = input.bool(true, "عرض مناطق الدعم والمقاومة", group=group_ui)
show_struct = input.bool(true, "عرض خطوط الهيكل", group=group_ui)
show_fvg = input.bool(true, "عرض الفجوات الصحية", group=group_ui)

// ==========================================
// 2. محرك الزمن الحقيقي والتأخير
// ==========================================
int tf_secs = timeframe.in_seconds(timeframe.period)
int delay_bars = tf_secs >= 300 ? 1 : math.max(1, math.round(300 / tf_secs))
f_delayed_src(src, bars) => src[bars]

delayed_open = f_delayed_src(open, delay_bars)
delayed_close = f_delayed_src(close, delay_bars)
delayed_high = f_delayed_src(high, delay_bars)
delayed_low = f_delayed_src(low, delay_bars)
delayed_vol = f_delayed_src(volume, delay_bars)

// ==========================================
// 3. رادار الميكرز + اختلال الأوامر
// ==========================================
vol_ma = ta.sma(volume, 20)
vol_std = ta.stdev(volume, 20)
float dynamic_vol_surge = vol_ma + (nz(vol_std) * 1.5)
delayed_dyn_vol = f_delayed_src(dynamic_vol_surge, delay_bars)
bool volume_surge = delayed_vol > delayed_dyn_vol

cmf_denom = delayed_high - delayed_low == 0 ? 0.001 : delayed_high - delayed_low
buy_vol = delayed_vol * ((delayed_close - delayed_low) / cmf_denom)
sell_vol = delayed_vol - buy_vol
float delta_vol = buy_vol - sell_vol

bool imbalance_bull = buy_vol > sell_vol * 3 and delayed_close > delayed_open
bool maker_bull = (volume_surge and delta_vol > 0) or imbalance_bull
bool maker_bear = volume_surge and delta_vol < 0

// ==========================================
// 4. نظام السوق + القوة النسبية (RS)
// ==========================================
bench_close = benchmark_ticker != "" ? request.security(benchmark_ticker, timeframe.period, close[1], barmerge.gaps_off, barmerge.lookahead_off) : na
bench_sma20 = ta.sma(bench_close, 20)
bench_roc = ta.roc(bench_close, 14)
bench_daily_change = benchmark_ticker != "" ? request.security(benchmark_ticker, "D", ta.roc(close, 1)) : 0.0

string market_regime = "محايد"
bool can_trade_market = true
if not na(bench_close)
    if bench_close > bench_sma20 and bench_roc > 0
        market_regime := "🟢 صاعد (BULL)"
        can_trade_market := true
    else if bench_close < bench_sma20 and bench_roc > -2
        market_regime := "🔴 هابط (BEAR)"
        can_trade_market := false
    else if bench_roc < -2 or bench_daily_change < -0.5
        market_regime := "🩸 انهيار (CRASH)"
        can_trade_market := false
else
    market_regime := "محايد (لا توجد بيانات مؤشر)"

stock_roc20 = ta.roc(delayed_close, 20)
bench_roc20 = ta.roc(bench_close, 20)
float relative_strength = stock_roc20 - bench_roc20
bool rs_positive = relative_strength > 0 or na(bench_close)

// ==========================================
// 5. التحليل الأساسي والمالي (تجاوز الأخطاء تلقائياً)
// ==========================================
// استخدام ignore_invalid_symbol = true لمنع توقف الكود عند نقص البيانات المالية
float pe_ratio = request.financial(syminfo.tickerid, "PRICE_EARNINGS", "FY", barmerge.gaps_off, true)
float roe_val = request.financial(syminfo.tickerid, "RETURN_ON_EQUITY", "FY", barmerge.gaps_off, true)
float debt_to_equity = request.financial(syminfo.tickerid, "TOTAL_DEBT_TO_EQUITY", "FY", barmerge.gaps_off, true)

pe_sma = ta.sma(pe_ratio, 200) 
roe_sma = ta.sma(roe_val, 200)

bool stock_undervalued = not na(pe_ratio) and not na(pe_sma) and pe_ratio > 0 and pe_ratio < pe_sma
bool financial_healthy = not na(roe_val) and not na(debt_to_equity) and roe_val > 10 and debt_to_equity < 150

string fund_state = "لا توجد بيانات"
string fin_state = "لا توجد بيانات"
if not na(pe_ratio) and not na(roe_val) and not na(debt_to_equity)
    fund_state := stock_undervalued ? "رخيص 🟢" : "غالي 🔴"
    fin_state := financial_healthy ? "سليم 🟢" : "مخاطرة 🔴"

// ==========================================
// 6. محرك البيانات والفريم الأعلى + فلتر الامتداد
// ==========================================
daily_ema200 = request.security(syminfo.tickerid, "D", ta.ema(close, 200)[1], barmerge.gaps_off, barmerge.lookahead_off)
sma50 = ta.sma(close, 50)
ema20 = ta.ema(close, 20)
bool daily_trend_up = not na(daily_ema200) ? (close > daily_ema200) : true
bool above_sma50 = close > sma50

atr_val = ta.atr(14)
atr_delayed = f_delayed_src(atr_val, delay_bars)

float extension_atr = atr_delayed > 0 ? (delayed_close - ema20) / atr_delayed : 0
bool deeply_extended = extension_atr > 2.2

upDays = math.max(close - open, 0)
downDays = math.max(open - close, 0)
upRsi = ta.rsi(upDays, 3)
downRsi = ta.rsi(downDays, 3)
rsi_val = 100 - (100 / (1 + upRsi / downRsi))
percentRank = ta.percentrank(ta.change(close), 100)
crsi = math.avg(rsi_val, rsi_val, percentRank)
float rsi_delayed = f_delayed_src(crsi, delay_bars)

float last_month_close = request.security(syminfo.tickerid, "1M", close[1], barmerge.gaps_off, barmerge.lookahead_off)
float monthly_return = last_month_close > 0 ? ((close - last_month_close) / last_month_close) * 100 : 0
bool pumped_retail = monthly_return > 25 and not maker_bull
bool low_volume_quality = delayed_vol < vol_ma * 0.8

// ==========================================
// 7. مصفوفات الدعوم والمقاومة الذكية
// ==========================================
pivot_h = ta.pivothigh(high, 5, 5)
pivot_l = ta.pivotlow(low, 5, 5)

var float[] res_levels = array.new_float()
var float[] sup_levels = array.new_float()

if not na(pivot_h)
    array.unshift(res_levels, pivot_h)
if not na(pivot_l)
    array.unshift(sup_levels, pivot_l)

if array.size(res_levels) > 0
    for i = array.size(res_levels) - 1 to 0
        if close > array.get(res_levels, i)
            array.remove(res_levels, i)

if array.size(sup_levels) > 0
    for i = array.size(sup_levels) - 1 to 0
        if close < array.get(sup_levels, i)
            array.remove(sup_levels, i)

float nearest_resist = na
float nearest_support = na

if array.size(res_levels) > 0
    float min_dist = 1e10
    for i = 0 to array.size(res_levels) - 1
        float lvl = array.get(res_levels, i)
        if lvl > close and (lvl - close) < min_dist
            min_dist := lvl - close
            nearest_resist := lvl

if array.size(sup_levels) > 0
    float min_dist = 1e10
    for i = 0 to array.size(sup_levels) - 1
        float lvl = array.get(sup_levels, i)
        if lvl < close and (close - lvl) < min_dist
            min_dist := close - lvl
            nearest_support := lvl

float zone_width = atr_delayed * 0.35
plot(show_zones and not na(nearest_resist) ? nearest_resist + zone_width : na, title="Resist High", color=color.new(color.red, 70), style=plot.style_linebr)
plot(show_zones and not na(nearest_resist) ? nearest_resist - zone_width : na, title="Resist Low", color=color.new(color.red, 70), style=plot.style_linebr)
plot(show_zones and not na(nearest_support) ? nearest_support + zone_width : na, title="Sup High", color=color.new(color.green, 70), style=plot.style_linebr)
plot(show_zones and not na(nearest_support) ? nearest_support - zone_width : na, title="Sup Low", color=color.new(color.green, 70), style=plot.style_linebr)

// ==========================================
// 8. بنية السوق (BOS / ChoCH)
// ==========================================
bool bos_bull = not na(nearest_resist) and ta.crossover(delayed_close, nearest_resist)
bool choch_bear = not na(nearest_support) and ta.crossunder(delayed_close, nearest_support)

var int trend_state = 0
if bos_bull
    trend_state := 1
    if show_struct
        label.new(bar_index, nearest_resist, "BOS", color=color.new(color.lime, 50), textcolor=color.white, style=label.style_label_down, size=size.tiny)
if choch_bear
    trend_state := -1
    if show_struct
        label.new(bar_index, nearest_support, "ChoCH", color=color.new(color.red, 50), textcolor=color.white, style=label.style_label_up, size=size.tiny)

bool sweep_low = not na(nearest_support) and delayed_low < nearest_support and delayed_close > nearest_support and delta_vol > 0
bool maker_sweep = sweep_low

var float active_ob = na
if (bos_bull or sweep_low) and show_zones and trend_state == 1
    active_ob := nearest_support
    box.new(bar_index[5 + delay_bars], high[5 + delay_bars], bar_index + 8, nearest_support, bgcolor=color.new(color.blue, 80), border_color=color.blue)

if choch_bear
    active_ob := na

// ==========================================
// 9. فحص صحة الفجوات (FVG Health Check)
// ==========================================
var float active_fvg_top = na
var float active_fvg_bot = na
bool is_bullish_fvg = delayed_low[0] > delayed_high[2]
if is_bullish_fvg
    active_fvg_top := delayed_low[0]
    active_fvg_bot := delayed_high[2]
    float fvg_size = active_fvg_top - active_fvg_bot
    bool fvg_healthy = fvg_size >= atr_delayed * 0.10 and fvg_size <= atr_delayed * 2.0
    if show_fvg and fvg_healthy
        box.new(bar_index[2], active_fvg_top, bar_index + 10, active_fvg_bot, bgcolor=color.new(color.green, 90), border_color=color.new(color.green, 50))

bool bull_trap = bos_bull and delta_vol < 0 and volume_surge
var int bull_trap_bar = na
if bull_trap
    bull_trap_bar := bar_index
bool block_after_trap = not na(bull_trap_bar) and (bar_index - bull_trap_bar) <= 5

plotshape(bull_trap, title="فخ شراء", style=shape.xcross, location=location.abovebar, color=color.red, size=size.small, text="فخ شراء 🚨")
plotshape(sweep_low, title="فخ بيع", style=shape.circle, location=location.belowbar, color=color.lime, size=size.small, text="فخ بيع 🎣")

// ==========================================
// 10. تتبع الصفقات + نظام الخروج الجزئي (Runner)
// ==========================================
var int total_trades = 0
var int winning_trades = 0
var int consecutive_losses = 0 
var float active_entry = na
var float active_tp1 = na
var float active_tp2 = na
var float active_sl = na
var int entry_bar = na
var float mfe = 0.0
var bool breakeven_active = false 
var bool partial_taken = false

if not na(active_entry)
    mfe := math.max(mfe, high)
    float current_r = active_entry - active_sl 
    
    if current_r > 0
        if high >= active_entry + (current_r * 1.5) and not partial_taken
            partial_taken := true
            breakeven_active := true
            active_sl := active_entry 
            
        if partial_taken
            float r_mult = (mfe - active_entry) / current_r
            float new_trail_sl = active_entry + (current_r * (r_mult - 1.0))
            if new_trail_sl > active_sl
                active_sl := new_trail_sl

    bool force_exit_runner = choch_bear or maker_bear
    
    if high >= active_tp2 or low <= active_sl or (partial_taken and force_exit_runner)
        if high >= active_tp2 or (partial_taken and high > active_entry)
            winning_trades += 1
            consecutive_losses := 0 
        else if not partial_taken
            consecutive_losses += 1
            
        total_trades += 1
        active_entry := na, active_tp1 := na, active_tp2 := na, active_sl := na, breakeven_active := false, partial_taken := false

float win_rate = total_trades > 0 ? (winning_trades / total_trades) * 100 : 0.0
bool safe_mode_active = consecutive_losses >= 3

float entry = delayed_close
float min_sl_dist = atr_delayed * 1.2
float structural_sl = not na(active_ob) ? active_ob - (atr_delayed * 0.5) : entry - min_sl_dist
float sl = math.min(structural_sl, entry - min_sl_dist) 
if sl >= entry
    sl := entry - min_sl_dist
float risk = entry - sl

float tp1 = not na(nearest_resist) and nearest_resist > entry ? nearest_resist : entry + (risk * 1.5)
float tp2 = entry + (risk * 4.0)

bool valid_rr = risk > 0 and ((tp1 - entry) / risk) >= 1.5

float size_multiplier = 1.0
if consecutive_losses >= 2
    size_multiplier := 0.5
if safe_mode_active
    size_multiplier := 0.0 

float risk_amount = account_balance * (risk_per_trade / 100) * size_multiplier
float position_size = risk > 0 ? risk_amount / risk : 0

string early_exit_warn = "لا يوجد"
if not na(active_entry)
    int bars_held = bar_index - entry_bar
    float mfe_r = risk > 0 ? (mfe - active_entry) / risk : 0
    if bars_held >= 3 and close < active_entry and mfe_r < 0.5 and maker_bear
        early_exit_warn := "⚠️ خروج مبكر (ركود وتصريف)"

// ==========================================
// 11. آلة الحالات الموحدة (The Holistic Quant Logic)
// ==========================================
string signal = "انتظر"
string path_taken = "مفيش دعم واضح أو R/R ضعيف"
color sig_color = color.gray

// لو البيانات المالية موجودة ومش سليمة بنرفض. لو مش موجودة بنعدّي.
bool fundamental_block = not na(debt_to_equity) and not financial_healthy

if safe_mode_active
    signal := "🚨 وقفة طوارئ (3 خسائر متتالية)"
    path_taken := "النظام فقد 3 صفقات ورا بعض"
    sig_color := color.red
else if fundamental_block
    signal := "🚫 ممنوع (أساسيات ضعيفة)"
    path_taken := "ديون عالية أو خسارة (ROE ضعيف)"
    sig_color := color.red
else if not can_trade_market or not daily_trend_up or not above_sma50 or trend_state == -1
    signal := "🚫 ممنوع دخول"
    path_taken := "سوق هابط أو هيكل مكسور (ChoCH)"
    sig_color := color.red
else if deeply_extended
    signal := "⏳ انتظر تصحيح (مطارد)"
    path_taken := "السهم ممتد جداً عن المتوسط"
    sig_color := color.orange
else if block_after_trap
    signal := "🚫 ممنوع دخول (فخ شراء)"
    path_taken := "تم رصد فخ شراء منذ قريب"
    sig_color := color.red
else if pumped_retail or low_volume_quality
    signal := "⚠️ متطاردش (جودة ضعيفة)"
    path_taken := "مضخة ناس عادية أو سيولة ضعيفة"
    sig_color := color.orange
else if not rs_positive
    signal := "انتظر"
    path_taken := "السهم أضعف من السوق (RS سلبي)"
    sig_color := color.gray
else if maker_sweep and valid_rr and maker_bull and rsi_delayed < 70 and trend_state == 1 and (financial_healthy or na(debt_to_equity))
    signal := "🟢 A+ Quant Setup (شراء)"
    path_taken := "فني قوي + R/R ممتاز"
    sig_color := color.purple
else if maker_bull and valid_rr and not na(active_ob) and rsi_delayed < 70 and trend_state == 1
    signal := "🟢 دخول محتمل"
    path_taken := "ميكرز بيشتري + OB صالح + R/R ≥ 1.5"
    sig_color := color.lime

if (signal == "🟢 A+ Quant Setup (شراء)" or signal == "🟢 دخول محتمل") and na(active_entry)
    active_entry := entry
    active_tp1 := tp1
    active_tp2 := tp2
    active_sl := sl
    entry_bar := bar_index
    mfe := high
    breakeven_active := false
    partial_taken := false

// ==========================================
// 12. لوحة القرار الموحدة (Ultimate Transparency Dashboard)
// ==========================================
var table dash = table.new(position.top_right, 4, 22, bgcolor=color.rgb(18, 22, 32), border_width=1, border_color=color.gray)

if barstate.islast
    table.cell(dash, 0, 0, syminfo.ticker + " | V56 Ultimate Quant", text_color=color.white, bgcolor=color.rgb(0, 80, 120), text_halign=text.align_center)
    table.merge_cells(dash, 0, 0, 3, 0)
    
    table.cell(dash, 0, 1, "الحالة:", text_color=color.gray)
    table.cell(dash, 1, 1, path_taken, text_color=color.white)
    table.cell(dash, 2, 1, "القرار:", text_color=color.gray)
    table.cell(dash, 3, 1, signal, text_color=sig_color)
    
    // قسم التحليل المالي والأساسي (بالأرقام الخام)
    table.cell(dash, 0, 2, "--- 🏦 التحليل المالي والأرقام ---", text_color=color.yellow, bgcolor=color.rgb(30, 35, 50))
    table.merge_cells(dash, 0, 2, 3, 2)

    table.cell(dash, 0, 3, "مكرر الربحية (P/E):", text_color=color.gray)
    table.cell(dash, 1, 3, not na(pe_ratio) ? str.tostring(pe_ratio, "#.#") + " (" + fund_state + ")" : "لا توجد بيانات", text_color=na(pe_ratio) ? color.gray : (stock_undervalued ? color.lime : color.white))
    table.cell(dash, 2, 3, "العائد (ROE):", text_color=color.gray)
    table.cell(dash, 3, 3, not na(roe_val) ? str.tostring(roe_val, "#.#") + "%" : "لا توجد بيانات", text_color=na(roe_val) ? color.gray : (roe_val > 10 ? color.lime : color.red))

    table.cell(dash, 0, 4, "الديون (D/E):", text_color=color.gray)
    table.cell(dash, 1, 4, not na(debt_to_equity) ? str.tostring(debt_to_equity, "#.#") + "%" : "لا توجد بيانات", text_color=na(debt_to_equity) ? color.gray : (debt_to_equity < 150 ? color.lime : color.red))
    table.cell(dash, 2, 4, "الصحة المالية:", text_color=color.gray)
    table.cell(dash, 3, 4, not na(debt_to_equity) ? fin_state : "لا توجد بيانات", text_color=na(debt_to_equity) ? color.gray : (financial_healthy ? color.lime : color.red))

    // قسم التحليل الفني
    table.cell(dash, 0, 5, "--- 📊 التحليل الفني والسيولة ---", text_color=color.yellow, bgcolor=color.rgb(30, 35, 50))
    table.merge_cells(dash, 0, 5, 3, 5)

    table.cell(dash, 0, 6, "نظام السوق:", text_color=color.gray)
    table.cell(dash, 1, 6, market_regime, text_color=can_trade_market ? color.lime : color.red)
    table.cell(dash, 2, 6, "حالة الهيكل:", text_color=color.gray)
    table.cell(dash, 3, 6, trend_state == 1 ? "🟢 صاعد (BOS)" : "🔴 هابط (ChoCH)", text_color=trend_state == 1 ? color.lime : color.red)

    table.cell(dash, 0, 7, "فوق المتوسطات:", text_color=color.gray)
    table.cell(dash, 1, 7, above_sma50 ? "✅ نعم" : "❌ لا", text_color=above_sma50 ? color.lime : color.red)
    table.cell(dash, 2, 7, "عند دعم/OB:", text_color=color.gray)
    table.cell(dash, 3, 7, not na(active_ob) ? "✅ صالح" : "❌ ملغي", text_color=not na(active_ob) ? color.lime : color.red)

    table.cell(dash, 0, 8, "الامتداد (ATR):", text_color=color.gray)
    table.cell(dash, 1, 8, str.tostring(extension_atr, "#.#"), text_color=deeply_extended ? color.red : color.white)
    table.cell(dash, 2, 8, "أداء الشهر:", text_color=color.gray)
    table.cell(dash, 3, 8, str.tostring(monthly_return, "#.#") + "%", text_color=pumped_retail ? color.red : color.white)

    table.cell(dash, 0, 9, "Liquidity Sweep:", text_color=color.gray)
    table.cell(dash, 1, 9, maker_sweep ? "✅ تم" : "❌ لا", text_color=maker_sweep ? color.lime : color.gray)
    table.cell(dash, 2, 9, "تدفق الأموال:", text_color=color.gray)
    table.cell(dash, 3, 9, imbalance_bull ? "🟢 سحب أوامر" : maker_bull ? "🟢 ميكرز بيشتري" : maker_bear ? "🔴 ميكرز بيبيع" : "⚪ ناس عادية", text_color=maker_bull ? color.lime : maker_bear ? color.red : color.gray)

    table.cell(dash, 0, 10, "نجاح النموذج:", text_color=color.gray)
    table.cell(dash, 1, 10, str.tostring(win_rate, "#.#") + "%", text_color=win_rate >= 60 ? color.lime : color.orange)
    table.cell(dash, 2, 10, "خسائر متتالية:", text_color=color.gray)
    table.cell(dash, 3, 10, str.tostring(consecutive_losses), text_color=consecutive_losses >= 3 ? color.red : color.white)

    // قسم خطة التداول
    table.cell(dash, 0, 11, "--- 💰 خطة التداول (Partial & Runner) ---", text_color=color.yellow, bgcolor=color.rgb(30, 35, 50))
    table.merge_cells(dash, 0, 11, 3, 11)

    table.cell(dash, 0, 12, "سعر الدخول:", text_color=color.gray)
    table.cell(dash, 1, 12, str.tostring(entry, "#.##"), text_color=color.white)
    table.cell(dash, 2, 12, "وقف الخسارة الحالي:", text_color=color.gray)
    table.cell(dash, 3, 12, not na(active_entry) ? str.tostring(active_sl, "#.##") + (partial_taken ? " (مربوط ✅)" : "") : str.tostring(sl, "#.##"), text_color=partial_taken ? color.lime : color.red)

    table.cell(dash, 0, 13, "الهدف الأول (50%):", text_color=color.gray)
